cmake_minimum_required(VERSION 3.24)
project(cs2_project VERSION 0.4.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_library(profile_core src/profile.cpp)
target_include_directories(profile_core PUBLIC src)
add_library(catalog_core src/catalog.cpp)
target_include_directories(catalog_core PUBLIC src)
add_library(bridge_core src/bridge_core.cpp)
target_link_libraries(bridge_core PUBLIC catalog_core)
if(MSVC)
  target_compile_options(profile_core PRIVATE /W4 /permissive- /utf-8)
  target_compile_options(catalog_core PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(profile_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(profile_tests tests/profile_tests.cpp)
  target_link_libraries(profile_tests PRIVATE profile_core)
  if(MSVC)
    target_compile_options(profile_tests PRIVATE /utf-8)
  endif()
  foreach(case roundtrip legacy malformed validation filesystem)
    add_test(NAME profile_${case} COMMAND profile_tests ${case})
  endforeach()
  add_executable(catalog_tests tests/catalog_tests.cpp)
  target_link_libraries(catalog_tests PRIVATE catalog_core)
  add_test(NAME catalog_mapping_and_search COMMAND catalog_tests)
  add_executable(bridge_tests tests/bridge_tests.cpp)
  target_link_libraries(bridge_tests PRIVATE bridge_core)
  add_test(NAME bridge_request_and_attribute_validation COMMAND bridge_tests)
endif()

option(BUILD_DESKTOP "Build the Windows desktop profile editor" ${WIN32})
set(PREVIEW_IMAGE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/assets/previews" CACHE PATH "Directory containing prepared preview PNGs")
get_filename_component(PREVIEW_IMAGE_DIR "${PREVIEW_IMAGE_DIR}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
if(BUILD_DESKTOP)
  if(NOT WIN32)
    message(FATAL_ERROR "Desktop requires Windows or a Windows cross compiler")
  endif()
  include(FetchContent)
  FetchContent_Declare(imgui
    GIT_REPOSITORY https://github.com/ocornut/imgui.git
    GIT_TAG f5befd2d29e66809cd1110a152e375a7f1981f06
  )
  FetchContent_MakeAvailable(imgui)
  add_library(imgui STATIC
    ${imgui_SOURCE_DIR}/imgui.cpp
    ${imgui_SOURCE_DIR}/imgui_draw.cpp
    ${imgui_SOURCE_DIR}/imgui_tables.cpp
    ${imgui_SOURCE_DIR}/imgui_widgets.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_win32.cpp
    ${imgui_SOURCE_DIR}/backends/imgui_impl_dx11.cpp
  )
  target_include_directories(imgui PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
  target_compile_definitions(imgui PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(imgui PUBLIC d3d11 dxgi dwmapi)
  if(MINGW)
    target_link_libraries(imgui PUBLIC d3dcompiler_47)
  else()
    target_link_libraries(imgui PUBLIC d3dcompiler)
  endif()
  add_library(cs2_bridge SHARED src/bridge_module_win32.cpp)
  set_target_properties(cs2_bridge PROPERTIES PREFIX "")
  target_compile_definitions(cs2_bridge PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(cs2_bridge PRIVATE bridge_core bcrypt)
  if(MINGW)
    # The supplied Zig/Clang toolchain supports Windows SEH with MS extensions.
    target_compile_options(cs2_bridge PRIVATE -fms-extensions)
  endif()
  add_executable(cs2_profiles WIN32 src/main_win32.cpp src/bridge_client_win32.cpp)
  add_library(image_decoder src/image_decode_win32.cpp)
  target_include_directories(image_decoder PUBLIC src)
  target_link_libraries(image_decoder PUBLIC windowscodecs ole32 uuid)
  target_link_libraries(cs2_profiles PRIVATE profile_core catalog_core bridge_core image_decoder imgui shell32 ole32 uuid bcrypt)
  if(EXISTS "${PREVIEW_IMAGE_DIR}")
    add_custom_command(TARGET cs2_profiles POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_directory "${PREVIEW_IMAGE_DIR}" "$<TARGET_FILE_DIR:cs2_profiles>/previews"
    )
  endif()
  if(BUILD_TESTING)
    add_executable(preview_tests tests/preview_tests_win32.cpp)
    target_link_libraries(preview_tests PRIVATE catalog_core image_decoder)
    if(MSVC)
      target_compile_options(preview_tests PRIVATE /utf-8)
    endif()
    add_test(NAME preview_decode_catalog COMMAND preview_tests "${PREVIEW_IMAGE_DIR}")
    add_test(NAME preview_window_smoke COMMAND cs2_profiles --preview-smoke-test)
    set_tests_properties(preview_window_smoke PROPERTIES TIMEOUT 30)
    add_executable(bridge_windows_tests tests/bridge_tests_win32.cpp)
    target_compile_definitions(bridge_windows_tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(bridge_windows_tests PRIVATE bridge_core bcrypt)
    add_test(NAME bridge_windows_reject_unsupported_host COMMAND bridge_windows_tests "$<TARGET_FILE:cs2_bridge>")
    set_tests_properties(bridge_windows_reject_unsupported_host PROPERTIES TIMEOUT 15)
  endif()
  if(MINGW)
    target_link_options(cs2_profiles PRIVATE -Wl,--subsystem,windows)
  endif()
  if(MSVC)
    target_compile_options(cs2_bridge PRIVATE /W4 /permissive- /utf-8)
    target_compile_options(cs2_profiles PRIVATE /W4 /permissive- /utf-8)
  endif()
endif()
